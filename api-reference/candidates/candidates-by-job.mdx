---

## Overview

Retrieve every candidate attached to a single job, sorted server-side. The defaults (`sortBy: RESUME_SCORE`, `sortDir: DESC`) return the job's roster stack-ranked out of the box; candidates without a `resumeScore` always sort after scored ones regardless of direction, so partially scored pools surface the scored rows first. For a paginated, cross-job list use `candidates`; for a paged version of this same stack rank use `jobStackRank`.

## Endpoint

```
POST /graphql
Content-Type: application/json
```

## GraphQL Query

```graphql
query CandidatesByJob(
  $jobId: UUID!
  $sortBy: CandidateSortField = RESUME_SCORE
  $sortDir: SortDirection = DESC
) {
  candidatesByJob(jobId: $jobId, sortBy: $sortBy, sortDir: $sortDir) {
    id
    name
    email
    resumeScore
    aiInterviewStatus
    codeAssessmentStatus
    codeAssessmentScore
    candidateStatus
    eligibility
    verdictId
    source
    isStarred
    createdAt
  }
}
```

## Request Parameters

<ParamField body="jobId" type="string" required>
  Unique identifier (UUID) of the job whose candidates to list.
</ParamField>

<ParamField body="sortBy" type="CandidateSortField" default="RESUME_SCORE">
  Server-side sort field. One of `RESUME_SCORE`, `CREATED_AT`, `NAME`. The default `RESUME_SCORE` powers the stack-rank table.
</ParamField>

<ParamField body="sortDir" type="SortDirection" default="DESC">
  Sort direction. One of `ASC`, `DESC`.
</ParamField>

## Response

<ResponseField name="data" type="object">
  <Expandable title="data properties">
    <ResponseField name="candidatesByJob" type="array">
      Array of candidate records for the job, in the requested sort order.

      <Expandable title="Candidate Object">
        <ResponseField name="id" type="string">
          Unique candidate identifier (UUID).
        </ResponseField>

        <ResponseField name="name" type="string">
          Candidate's full name.
        </ResponseField>

        <ResponseField name="email" type="string">
          Email address, if known.
        </ResponseField>

        <ResponseField name="resumeScore" type="number">
          Aggregate AI resume score. Null when the resume has not been scored; unscored candidates sort after scored ones.
        </ResponseField>

        <ResponseField name="aiInterviewStatus" type="string">
          Status of the candidate's latest AI interview, if any.
        </ResponseField>

        <ResponseField name="codeAssessmentStatus" type="string">
          Denormalized status of the latest coding assessment invite for this candidate. Set to `Scheduled` when the assessment is sent and refreshed to `Completed` or `Expired` on finalize.
        </ResponseField>

        <ResponseField name="codeAssessmentScore" type="number">
          Score from the latest completed coding assessment, if any.
        </ResponseField>

        <ResponseField name="candidateStatus" type="string">
          Lifecycle status in the hiring pipeline. One of `APPLIED`, `IN_PROGRESS`, `ADVANCED`, `REJECTED`, `HIRED`, `WITHDRAWN`.
        </ResponseField>

        <ResponseField name="eligibility" type="string">
          Recruiter's eligibility decision. One of `ELIGIBLE`, `INELIGIBLE`, `UNSET`. Legacy rows surface as `UNSET`.
        </ResponseField>

        <ResponseField name="verdictId" type="string">
          Recruiter or AI verdict slug (e.g. `shortlisted`, `ai_screening`). Distinct from the `candidateStatus` lifecycle; missing on legacy rows.
        </ResponseField>

        <ResponseField name="source" type="string">
          Provenance of the candidate row. One of `MANUAL`, `BULK_UPLOAD`, `INTERNAL_POOL`, `CAREERS_PAGE`.
        </ResponseField>

        <ResponseField name="isStarred" type="boolean">
          Recruiter-marked standout candidate.
        </ResponseField>

        <ResponseField name="createdAt" type="string">
          ISO 8601 timestamp of creation.
        </ResponseField>
      </Expandable>
    </ResponseField>
  </Expandable>
</ResponseField>
